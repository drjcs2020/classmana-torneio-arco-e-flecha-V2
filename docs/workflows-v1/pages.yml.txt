name: Publicar ClassMana Torneio no GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: github-pages
  cancel-in-progress: false

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Baixar código
        uses: actions/checkout@v4
      - name: Extrair kit de artes aprovado (quando presente)
        run: |
          if [ -f ClassMana_Integracao_Visual_01.zip ]; then
            unzip -q -o ClassMana_Integracao_Visual_01.zip -d .
            test -f assets/ui/hud/forca-01-moldura.png
            test -f assets/scenes/aguia.webp
            echo "Sprites individuais e cenários disponíveis."
          else
            echo "Kit ainda não enviado: jogo existente permanece funcional com arte antiga."
          fi
      - name: Descompactar partes 1 e 2 do novo kit ClassMana
        run: |
          P1="ClassMana_Assets_Parte_1_Personagens_Interface.zip"
          P2="ClassMana_Assets_Parte_2_Cenarios_Botoes_HUD.zip"
          if [ -f "$P1" ] && [ -f "$P2" ]; then
            unzip -q -o "$P1" -d .
            unzip -q -o "$P2" -d .
            test -f assets/ui/buttons/botao-jogar-3d.png
            test -f assets/ui/hud/forca-01-moldura.png
            test -f assets/ui/hud/logo-torneio.png
            test -f assets/ui/hud/brasao-aguia-dourada.png
            test -f assets/scenes/aguia.webp
            test -f assets/scenes/leao.webp
            test -f assets/scenes/dragao.webp
            echo "OK: kit dividido em dois ZIPs publicado."
          elif [ -f "$P1" ] || [ -f "$P2" ]; then
            echo "::warning::Apenas uma parte do kit foi enviada. Envie os dois ZIPs para ativar as novas artes."
          else
            echo "As novas duas partes ainda não estão presentes no repositório."
          fi
      - name: Configurar GitHub Pages
        uses: actions/configure-pages@v5
      - name: Montar site público sem ZIPs e código privado
        run: |
          mkdir -p _site
          for file in *.html *.css *.js .nojekyll; do
            if [ -f "$file" ]; then cp -- "$file" _site/; fi
          done
          cp -a assets _site/assets
          test -s _site/index.html
          test -s _site/online-client.js
          test -s _site/assets/scenes/aguia.webp
          echo "Site estático preparado: $(du -sh _site | cut -f1); ZIPs de 35 MB removidos da publicação."
      - name: Preparar arquivos do jogo
        uses: actions/upload-pages-artifact@v4
        with:
          path: '_site'
      - name: Publicar site
        id: deployment
        uses: actions/deploy-pages@v4
