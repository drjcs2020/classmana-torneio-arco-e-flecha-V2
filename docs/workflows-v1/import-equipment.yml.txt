name: Importar recursos de equipamento 3D ClassMana
on:
  push:
    branches: [main]
    paths: ['.github/workflows/import-equipment.yml']
  workflow_dispatch:
permissions:
  contents: write
jobs:
  importar:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Copiar ícones da biblioteca ClassMana
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p assets/equipment
          base="https://raw.githubusercontent.com/drjcs2020/classmana-batalha-dos-monstros-V2/main/assets/items3d"
          for asset in bow compass gem shield torch scroll crystal rocks; do
            if [ ! -s "assets/equipment/$asset.png" ]; then
              curl --fail --silent --show-error --retry 3 --location "$base/$asset.png" -o "assets/equipment/$asset.png"
            fi
          done
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add assets/equipment
          if ! git diff --cached --quiet; then
            git commit -m "assets: adicionar ícones 3D originais de equipamento ClassMana"
            git push
          fi
